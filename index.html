            const res = await fetch(searchUrl);

                        .filter(l => l.length > 10 && !l.startsWith('=='));

            .replace(/[^A-Z]/g, '');

    };

    async function loadFullOdsDictionary() {

        const urls = [

            'https://raw.githubusercontent.com/chrplr/openlexicon/master/datasets-info/Liste-de-mots-francais-Gutenberg/liste.de.mots.francais.frgut.txt',

            'https://raw.githubusercontent.com/Taknok/French-Wordlist/master/francais.txt'

        ];

        for (const url of urls) {

            try {

                const res = await fetch(url);

                if (!res.ok) continue;

                const txt = await res.text();

                const lines = txt.split(/\r?\n/);

                if (lines.length > 10000) {

                    for (let i = 0; i < lines.length; i++) {

                        const norm = normalizeOdsWord(lines[i]);

                        if (norm.length >= 2 && norm.length <= 15) {

                            FULL_ODS_SET.add(norm);

                        }

                    }

                    break;

                }

            } catch (e) {}

        }

    }


    async function isWordValidOds(rawWord) {

        const norm = normalizeOdsWord(rawWord);

        if (!norm || norm.length < 2 || norm.length > 15) return false;

        if (FULL_ODS_SET.has(norm)) return true;


        // Repli en ligne via Wiktionnaire / recherche insensible aux accents

        try {

            const lower = norm.toLowerCase();

            const res = await fetch(`https://fr.wiktionary.org/w/api.php?action=opensearch&search=${encodeURIComponent(lower)}&limit=5&namespace=0&format=json&origin=*`);

            if (res.ok) {

                const data = await res.json();

                const titles = Array.isArray(data[1]) ? data[1] : [];

                if (titles.some(t => normalizeOdsWord(t) === norm)) {

                    FULL_ODS_SET.add(norm);

                    return true;

                }

            }

        } catch (e) {}

        return false;

    }

    function playWoodSound(freq = 420, duration = 0.08) {

        if (!soundEnabled) return;

        try {

            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = 'triangle';

            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            osc.frequency.exponentialRampToValueAtTime(freq * 0.55, audioCtx.currentTime + duration);

            gain.gain.setValueAtTime(0.14, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + duration);

        } catch (e) {}

    }

    function createShuffledBag() {

        const bag = [];

        TILE_DISTRIBUTION.forEach(item => {

            for (let i = 0; i < item.count; i++) {

                bag.push({ letter: item.letter, points: item.points, isJoker: item.letter === '?' });

            }

        });

        for (let i = bag.length - 1; i > 0; i--) {

            const j = Math.floor(Math.random() * (i + 1));

            [bag[i], bag[j]] = [bag[j], bag[i]];

        }

        return bag;

    }


    function drawTiles(bag, count) {

        const drawn = [];

        for (let i = 0; i < count && bag.length > 0; i++) {

            drawn.push(bag.pop());

        }

        return drawn;

    }

    window.abandonRecentGame = function(gameId) {

        const list = getStoredRecentGames();

        const idx = list.findIndex(g => g.id === gameId);

        if (idx === -1 || !list[idx].state) return;


        const st = list[idx].state;

        const userName = getCurrentUserDisplayName();

        let pIdx = st.players.findIndex(p => p.name.toLowerCase() === userName.toLowerCase());

        if (pIdx === -1) pIdx = 0;


        const leavingPlayer = st.players[pIdx];

        if (leavingPlayer && Array.isArray(leavingPlayer.rack)) {

            st.bag.push(...leavingPlayer.rack);

        }

        st.players.splice(pIdx, 1);


        if (st.players.length <= 1) {

            st.gameOver = true;

            list[idx].gameOver = true;

        } else {

            st.currentPlayerIndex = st.currentPlayerIndex % st.players.length;

        }


        list[idx].players = st.players.map(p => ({ name: p.name, score: p.score, avatar: p.avatar, isAI: p.isAI }));

        list[idx].title = st.players.map(p => p.name).join(' vs ');

        saveStoredRecentGames(list);


        if (gameState.id === gameId) {

            gameState = st;

            tempPlacements = [];

            renderAll();

        }

        renderRecentGamesUI();

        showToast(`${leavingPlayer ? leavingPlayer.name : 'Le joueur'} a abandonné ; la partie continue sans lui.`);

    };

    function evaluateBoardMove() {

        if (tempPlacements.length === 0) return { valid: false, reason: 'Posez au moins une tuile.' };


        const rows = new Set(tempPlacements.map(t => t.row));

        const cols = new Set(tempPlacements.map(t => t.col));

        if (rows.size > 1 && cols.size > 1) {

            return { valid: false, reason: 'Alignez les tuiles sur une même ligne ou colonne.' };

        }


        const isHorizontal = rows.size === 1 && (tempPlacements.length > 1 || hasHorizontalNeighbor(tempPlacements[0].row, tempPlacements[0].col));

        const tempGrid = gameState.board.map(row => [...row]);

        tempPlacements.forEach(tp => {

            tempGrid[tp.row][tp.col] = tp;

        });


        const isBoardEmpty = gameState.board[7][7] === null;

        if (isBoardEmpty) {

            if (!tempPlacements.some(t => t.row === 7 && t.col === 7)) {

                return { valid: false, reason: 'Le premier mot doit passer sur l’étoile ★.' };

            }

            if (tempPlacements.length < 2) {

                return { valid: false, reason: 'Formez un mot d’au moins 2 lettres.' };

            }

        }


        // Vérifier continuité et contact avec le plateau existant

        let touchesExisting = isBoardEmpty;

        if (rows.size === 1) {

            const r = tempPlacements[0].row;

            const minC = Math.min(...tempPlacements.map(t => t.col));

            const maxC = Math.max(...tempPlacements.map(t => t.col));

            for (let c = minC; c <= maxC; c++) {

                if (!tempGrid[r][c]) return { valid: false, reason: 'Les tuiles doivent se toucher sans trou.' };

                if (gameState.board[r][c]) touchesExisting = true;

            }

        } else {

            const c = tempPlacements[0].col;

            const minR = Math.min(...tempPlacements.map(t => t.row));

            const maxR = Math.max(...tempPlacements.map(t => t.row));

            for (let r = minR; r <= maxR; r++) {

                if (!tempGrid[r][c]) return { valid: false, reason: 'Les tuiles doivent se toucher sans trou.' };

                if (gameState.board[r][c]) touchesExisting = true;

            }

        }


        if (!touchesExisting) {

            for (const tp of tempPlacements) {

                const adj = [[1,0],[-1,0],[0,1],[0,-1]];

                if (adj.some(([dr, dc]) => gameState.board[tp.row + dr]?.[tp.col + dc])) {

                    touchesExisting = true;

                    break;

                }

            }

        }

        if (!touchesExisting) {

            return { valid: false, reason: 'Accrochez votre mot à une lettre déjà posée.' };

        }


        const formedWords = [];

        const extractWord = (startR, startC, dr, dc) => {

            let r = startR, c = startC;

            while (tempGrid[r - dr]?.[c - dc]) { r -= dr; c -= dc; }

            let word = '';

            let wordScore = 0;

            let wordMult = 1;

            const coords = [];

            while (tempGrid[r]?.[c]) {

                const cellTile = tempGrid[r][c];

                const isNew = tempPlacements.some(tp => tp.row === r && tp.col === c);

                let letterScore = cellTile.points;

                if (isNew) {

                    const bonus = BOARD_LAYOUT[r][c];

                    if (bonus === 'DL') letterScore *= 2;

                    else if (bonus === 'TL') letterScore *= 3;

                    else if (bonus === 'DW' || bonus === 'STAR') wordMult *= 2;

                    else if (bonus === 'TW') wordMult *= 3;

                }

                word += cellTile.letter;

                wordScore += letterScore;

                coords.push({ row: r, col: c });

                r += dr;

                c += dc;

            }

            return { word, score: wordScore * wordMult, coords };

        };


        const mainWord = isHorizontal

            ? extractWord(tempPlacements[0].row, tempPlacements[0].col, 0, 1)

            : extractWord(tempPlacements[0].row, tempPlacements[0].col, 1, 0);


        if (mainWord.word.length >= 2) formedWords.push(mainWord);


        for (const tp of tempPlacements) {

            const cross = isHorizontal

                ? extractWord(tp.row, tp.col, 1, 0)

                : extractWord(tp.row, tp.col, 0, 1);

            if (cross.word.length >= 2) {

                if (!formedWords.some(fw => fw.word === cross.word && fw.coords[0].row === cross.coords[0].row && fw.coords[0].col === cross.coords[0].col)) {

                    formedWords.push(cross);

                }

            }

        }


        if (formedWords.length === 0) {

            return { valid: false, reason: 'Formez un mot d’au moins 2 lettres.' };

        }


        let totalScore = formedWords.reduce((acc, w) => acc + w.score, 0);

        if (tempPlacements.length === 7) totalScore += 50; // Scrabble !


        return { valid: true, words: formedWords, totalScore };

    }


    function hasHorizontalNeighbor(r, c) {

        return Boolean(gameState.board[r]?.[c - 1] || gameState.board[r]?.[c + 1]);

    }

    async function runAITurn(aiPlayer) {

        if (gameState.gameOver) return;

        const comp = COMPANIONS.find(c => c.id === aiPlayer.avatar) || COMPANIONS[0];

        updateMascotBubble(comp.id, comp.quote);


        const move = findBestAIMove(aiPlayer);

        if (move) {

            applyValidatedMove(aiPlayer, move.placements, move.evalRes);

        } else {

            if (gameState.bag.length >= 7) {

                const count = Math.min(3, aiPlayer.rack.length);

                const returned = aiPlayer.rack.splice(0, count);

                aiPlayer.rack.push(...drawTiles(gameState.bag, count));

                gameState.bag.push(...returned);

                showToast(`${aiPlayer.name} échange ${count} lettres.`);

            } else {

                gameState.consecutivePasses++;

                showToast(`${aiPlayer.name} passe son tour.`);

            }

            if (gameState.consecutivePasses >= gameState.players.length * 2) finishGame();

            else advanceTurn();

        }

    }


    function findBestAIMove(aiPlayer) {

        const isBoardEmpty = gameState.board[7][7] === null;

        const candidates = Array.from(BUILTIN_ODS_WORDS).filter(w => w.length >= 2 && w.length <= 6);

        // Mélanger quelques mots candidats pour varier le jeu

        for (let i = candidates.length - 1; i > 0; i--) {

            const j = Math.floor(Math.random() * (i + 1));

            [candidates[i], candidates[j]] = [candidates[j], candidates[i]];

        }

        const sample = candidates.slice(0, 260);


        if (isBoardEmpty) {

            for (const word of sample) {

                const built = tryBuildWordFromRack(word, aiPlayer.rack);

                if (built) {

                    const placements = built.map((tileObj, idx) => ({

                        row: 7,

                        col: 7 + idx,

                        letter: tileObj.chosenLetter,

                        isJoker: tileObj.tile.isJoker,

                        rackIndex: tileObj.rackIndex

                    }));

        const anchors = [];

        for (let r = 0; r < 15; r++) {

            for (let c = 0; c < 15; c++) {

                if (gameState.board[r][c]) anchors.push({ r, c, letter: gameState.board[r][c].letter });

            }

        }


        for (const anchor of anchors) {

            for (const word of sample) {

                const pos = word.indexOf(anchor.letter);

                if (pos === -1) continue;


                // Essai horizontal

                const startC = anchor.c - pos;

                if (startC >= 0 && startC + word.length <= 15) {

                    const needed = [];

                    let ok = true;

                    for (let i = 0; i < word.length; i++) {

                        const cc = startC + i;

                        const existing = gameState.board[anchor.r][cc];

                        if (existing) {

                            if (existing.letter !== word[i]) { ok = false; break; }

                        } else {

                            needed.push({ row: anchor.r, col: cc, letter: word[i] });

                        }

                    }

                    if (ok && needed.length > 0 && needed.length <= aiPlayer.rack.length) {

                        const built = tryBuildWordFromRack(needed.map(n => n.letter).join(''), aiPlayer.rack);

                        if (built) {

                            const placements = needed.map((n, idx) => ({

                                row: n.row,

                                col: n.col,

                                letter: built[idx].chosenLetter,

                                points: built[idx].tile.points,

                                isJoker: built[idx].tile.isJoker,

                                rackIndex: built[idx].rackIndex

                            }));

                            const evalRes = evaluateBoardMove();

                            tempPlacements = [];

                                return { placements, evalRes };

                            }

                        }

                    }

                }

            }

        }

        return null;

    }


    function tryBuildWordFromRack(neededLetters, rack) {

        const used = new Set();

        const result = [];

        for (const ch of neededLetters) {

            let idx = rack.findIndex((t, i) => !used.has(i) && !t.isJoker && t.letter === ch);

            if (idx === -1) {

                idx = rack.findIndex((t, i) => !used.has(i) && t.isJoker);

            }

            if (idx === -1) return null;

            used.add(idx);

            result.push({ tile: rack[idx], rackIndex: idx, chosenLetter: ch });

        }

        return result;

    }
